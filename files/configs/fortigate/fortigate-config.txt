3. DHCP
============================================================

Comando:
show system dhcp server

config system dhcp server
    edit 1
        set ntp-service local
        set default-gateway 10.255.1.1
        set netmask 255.255.255.0
        set interface "fortilink"
        config ip-range
            edit 1
                set start-ip 10.255.1.2
                set end-ip 10.255.1.254
            next
        end
        set vci-match enable
        set vci-string "FortiSwitch" "FortiExtender"
    next
    edit 2
        set dns-service default
        set default-gateway 10.21.75.1
        set netmask 255.255.255.128
        set interface "port2"
        config ip-range
            edit 1
                set start-ip 10.21.75.10
                set end-ip 10.21.75.100
            next
        end
    next
end

Comando:

config firewall policy
    edit 1
        set name "Usuarios_a_Internet"
        set uuid b54d1ec4-b7b0-51f1-21f8-2933c4ee3975
        set srcintf "port2"
        set dstintf "port1"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set utm-status enable
        set ssl-ssh-profile "deep-inspection"
        set file-filter-profile "Bloquear_EXE"
        set logtraffic all
        set nat enable
    next
    edit 2
        set name "Usuarios_a_WEB"
        set uuid 4f41dd44-b7b1-51f1-99e4-51171820e2ca
        set srcintf "port2"
        set dstintf "port3"
        set action accept
        set srcaddr "all"
        set dstaddr "WEB-SERVER"
        set schedule "always"
        set service "HTTPS" "HTTP"
        set utm-status enable
        set profile-protocol-options "default-1"
        set ssl-ssh-profile "custom-deep-inspection"
        set file-filter-profile "Bloquear_EXE"
        set ips-sensor "IPS_SQLi_Cuarentena"
        set logtraffic all
        set logtraffic-start enable
    next
    edit 3
        set name "Usuarios_a_DB_BLOQUEADO"
